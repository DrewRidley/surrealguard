DEFINE TABLE person SCHEMAFULL;
DEFINE FIELD name ON person TYPE string;
DEFINE TABLE account SCHEMAFULL;
DEFINE TABLE has_account SCHEMAFULL TYPE RELATION FROM person TO account;
CREATE person:p SET name = 'A';
CREATE account:a;
RELATE person:p->has_account->account:a;
SELECT *, name AS n FROM person;
SELECT *, ->has_account AS acc FROM person;
DEFINE TABLE ea SCHEMALESS;
RETURN IF [] { 'truthy' } ELSE { 'falsy' };
RETURN IF (SELECT count() FROM ea) { 'truthy' } ELSE { 'falsy' };
CREATE ea:1;
RETURN IF (SELECT count() FROM ea) { 'truthy' } ELSE { 'falsy' };
RETURN array::is_empty(SELECT count() FROM ea WHERE id = ea:99);
RETURN array::is_empty(SELECT count() FROM ea WHERE id = ea:99 GROUP ALL);
RETURN array::len(SELECT count() FROM ea);
